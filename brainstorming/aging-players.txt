def _initial_age(self) -> int:
    # Peak around 26–28
    age = int(self.rng.gauss(mu=27, sigma=4))
    return max(18, min(40, age))

class LeagueBootstrapper:
    def seed_players(self, num_players: int) -> list[PlayerRecord]:
        players = [self.player_factory.create() for _ in range(num_players)]

        for p in players:
            p.age = self._initial_age()
            p.experience = max(0, p.age - 18)

        return players
