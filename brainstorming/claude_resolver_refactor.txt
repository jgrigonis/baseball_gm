Extract Count Modifiers to Lookup Table
COUNT_MODIFIERS = {
    (0, 0): {"ball": 1.15, "strike": 1.33, "foul": 0.66, "home_run": 0.61, "in_play": 0.66},
    (1, 0): {"ball": 0.92, "strike": 1.12, "foul": 0.94, "home_run": 1.1, "in_play": 1.1},
    # ... all counts
}

def _modify_for_count(self, count: Count):
    mods = COUNT_MODIFIERS.get((count.balls, count.strikes))
    if mods:
        for outcome, factor in mods.items():
            self.pitch_outcome.multiply(outcome, factor)
